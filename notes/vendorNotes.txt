userSchema.pre('save', function (next) {
    if (this.isModified('role') && this.role === 'vendor') {
        this.status = 'active';
    } else {
        this.status = 'pending';
        this.shopName = undefined;
    }

    next();
});

            this.status = 'pending';

        }
    }

    next();
});
